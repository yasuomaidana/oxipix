//! Scratch benchmark: oxipix `imgproc` (main @ 412b476) on the rust-cv-guide fixtures, same timing rule as
//! comparisons/opencv/bench.cpp, output allocated per call (compare with OpenCV `__fresh`).
//! Usage: obench <assets dir> <opencv json>

use image::{GrayImage, ImageBuffer, Luma, Pixel, RgbImage};
use oxipix::imgproc::{
    arith, bilateral, blur, canny, equalize, features, filter2d, gradient, hough, median, morph, pyramid,
    quality, template, threshold, warp,
};
use std::hint::black_box;
use std::time::Instant;

fn bench_ms(iters: usize, mut f: impl FnMut()) -> (f64, f64) {
    let mut first = 0.0f64;
    for _ in 0..3 {
        let t = Instant::now();
        f();
        first = first.max(t.elapsed().as_secs_f64() * 1e3);
    }
    let wanted = ((200.0 / first.max(1e-6)) as usize).clamp(iters, 5000);
    let mut s: Vec<f64> = (0..wanted)
        .map(|_| {
            let t = Instant::now();
            f();
            t.elapsed().as_secs_f64() * 1e3
        })
        .collect();
    s.sort_by(|a, b| a.partial_cmp(b).unwrap());
    let med = s[s.len() / 2];
    let mut dev: Vec<f64> = s.iter().map(|v| (v - med).abs()).collect();
    dev.sort_by(|a, b| a.partial_cmp(b).unwrap());
    (med, dev[dev.len() / 2])
}

fn mean<P: Pixel<Subpixel = u8>>(im: &ImageBuffer<P, Vec<u8>>) -> Option<f64> {
    let r = im.as_raw();
    Some(r.iter().map(|&v| v as f64).sum::<f64>() / r.len().max(1) as f64)
}

fn count<T>(v: &[T]) -> Option<f64> {
    Some(v.len() as f64)
}

fn main() {
    let args: Vec<String> = std::env::args().collect();
    let dir = &args[1];
    let ocv: serde_json::Value = serde_json::from_str(&std::fs::read_to_string(&args[2]).unwrap()).unwrap();
    let p = |n: &str| format!("{dir}/{n}");
    let rgb = |n: &str| -> RgbImage { image::open(p(n)).unwrap().to_rgb8() };
    let gray = |n: &str| -> GrayImage { image::open(p(n)).unwrap().to_luma8() };
    let (chelsea, astronaut, invoice_photo) = (rgb("chelsea.png"), rgb("astronaut.png"), rgb("invoice_photo.png"));
    let (linux, win, coffee) = (rgb("LinuxLogo.jpg"), rgb("WindowsLogo.jpg"), rgb("coffee.png"));
    let (coins, building, chelsea_gray) = (gray("coins.png"), gray("building.jpg"), gray("chelsea.png"));
    let coins_mask = threshold::threshold_binary(&coins, threshold::otsu_level(&coins));
    let building_blur = blur::gaussian_blur_radius(&building, 2.0, 6);
    let mut chelsea_blur = blur::gaussian_blur_radius(&chelsea_gray, 1.5, 5);
    chelsea_blur.pixels_mut().for_each(|px| px.0[0] = px.0[0].saturating_add(25));
    let (tw, th) = (40, 40);
    let tmpl = image::imageops::crop_imm(&chelsea_gray, (chelsea_gray.width() - tw) / 2, chelsea_gray.height() * 2 / 5 - 20, tw, th)
        .to_image();

    let mut rows: Vec<(&str, f64, f64, Option<f64>)> = Vec::new();
    macro_rules! run {
        ($id:literal, $iters:expr, $body:expr, $dig:expr) => {{
            let f = || $body;
            let out = f();
            let d: Option<f64> = $dig(&out);
            let (ms, mad) = bench_ms($iters, || {
                black_box(f());
            });
            rows.push(($id, ms, mad, d));
        }};
    }


    run!("adding_images", 50, arith::add_weighted(&linux, &win, (0.5, 0.5, 0.0)).unwrap(), |o| mean(o));
    run!("basic_linear_transform", 50, arith::convert_scale(&coffee, 1.3, 40.0), |o| mean(o));
    run!("smoothing_box", 30, blur::box_blur(&chelsea, 2), |o| mean(o));
    run!("smoothing_gaussian", 30, blur::gaussian_blur_radius(&chelsea, 1.1, 2), |o| mean(o));
    run!("smoothing_median", 15, median::median_blur(&chelsea, 2), |o| mean(o));
    run!("smoothing_bilateral", 10, bilateral::bilateral_filter(&chelsea, 2, 30.0, 30.0), |o| mean(o));
    let box_k = [1.0f32 / 25.0; 25];
    run!("filter_2d", 30, filter2d::filter2d(&chelsea, &box_k, 5), |o| mean(o));
    run!("equalize", 50, equalize::equalize_hist(&building), |o| mean(o));
    run!("threshold", 50, threshold::threshold_binary(&coins, threshold::otsu_level(&coins)), |o| mean(o));
    run!("erosion", 50, morph::erode(&coins_mask, 2, 2), |o| mean(o));
    run!("opening_closing_hats", 30, morph::top_hat(&coins, 3, 3), |o| mean(o));
    run!("distance_transform", 30, oxipix::imgproc::distance::euclidean(&coins_mask), |_: &_| None);
    run!("canny", 30, canny::canny(&building, 50.0, 150.0), |o| mean(o));
    run!("sobel", 30, gradient::scale_abs(&gradient::sobel_x(&building)), |o| mean(o));
    let cp = hough::CircleParams { min_radius: 15, max_radius: 40, min_distance: 20.0, canny_high: 150.0, votes: 25 };
    run!("hough_circle", 15, hough::hough_circles(&coins, &cp), |o: &Vec<_>| count(o));
    run!("hough_lines", 20, hough::hough_lines(&canny::canny(&building, 50.0, 150.0), 150), |o: &Vec<_>| count(o));
    // Forward rotation as cv::getRotationMatrix2D(center, -30, 0.6), inverted to the destination-to-source map.
    let (cx, cy) = (astronaut.width() as f32 / 2.0, astronaut.height() as f32 / 2.0);
    let ang = (-30.0f32).to_radians();
    let (a, b) = (0.6 * ang.cos(), 0.6 * ang.sin());
    let fwd = [a, b, (1.0 - a) * cx - b * cy, -b, a, b * cx + (1.0 - a) * cy];
    let inv = warp::invert_affine(&fwd).unwrap();
    run!("warp_affine", 30, warp::warp_affine(&astronaut, &inv, astronaut.dimensions()), |o| mean(o));
    run!("pyramids", 30, pyramid::pyr_down(&pyramid::pyr_down(&pyramid::pyr_down(&chelsea))), |o| mean(o));
    run!("template_matching", 20, template::match_template(&chelsea_blur, &tmpl, template::Method::Sse), |_: &_| None);
    let src = [(70.0, 40.0), (580.0, 70.0), (555.0, 430.0), (40.0, 400.0)];
    let dst = [(0.0, 0.0), (640.0, 0.0), (640.0, 480.0), (0.0, 480.0)];
    run!("homography", 30, {
        let m = warp::homography(&dst, &src).unwrap();
        warp::warp_perspective(&invoice_photo, &m, invoice_photo.dimensions())
    }, |o| mean(o));
    run!("feature_detection_fast", 30, {
        let c = features::fast9(&building, 30);
        features::suppress_non_max(&c, building.width(), building.height())
    }, |o: &Vec<_>| count(o));
    run!("psnr", 50, quality::psnr(&building, &building_blur), |_: &_| None);
    run!("ssim", 20, quality::ssim(&building, &building_blur), |_: &_| None);
    let _ = Luma([0u8]);

    let res = &ocv["results"];
    let dig = &ocv["digests"];
    println!("{:24} {:>9} {:>7} {:>9} {:>7} {:>10} {:>10}", "op", "oxipix_ms", "mad", "opencv", "ratio", "digest", "ocv_dig");
    for (id, ms, mad, d) in &rows {
        let o = res[format!("{id}__fresh")].as_f64().unwrap_or(f64::NAN);
        let od = &dig[*id];
        let od = if od["h"].as_i64() == Some(1) { od["w"].as_f64() } else { od["mean"].as_f64() };
        println!(
            "{:24} {:9.4} {:7.4} {:9.4} {:7.2} {:>10} {:>10}",
            id,
            ms,
            mad,
            o,
            ms / o,
            d.map_or("-".into(), |m| format!("{m:.3}")),
            od.map_or("-".into(), |m| format!("{m:.3}"))
        );
    }
}
