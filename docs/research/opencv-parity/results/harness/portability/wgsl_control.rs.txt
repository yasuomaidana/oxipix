//! Control for the cubecl spike: oxipix's own CONVOLUTION_2D_WGSL, with buffers and the bind group created once.
//! "resident" = dispatch + wait; "pooled round trip" = write_buffer upload + dispatch + copy + map read, reusing buffers.
use std::time::Instant;
use wgpu::util::DeviceExt;

const LEAN_WGSL: &str = r#"
struct Dimensions { width: u32, height: u32, _p0: u32, _p1: u32 };
struct Kernel { weights: array<vec4<f32>, 3> };
@group(0) @binding(0) var<storage, read> input_rgba: array<u32>;
@group(0) @binding(1) var<uniform> kernel_weights: Kernel;
@group(0) @binding(2) var<uniform> dims: Dimensions;
@group(0) @binding(3) var<storage, read_write> output_rgba: array<u32>;
@compute @workgroup_size(16, 16)
fn main(@builtin(global_invocation_id) id: vec3<u32>) {
    let x = id.x; let y = id.y;
    if (x >= dims.width || y >= dims.height) { return; }
    var acc = vec3<f32>(0.0);
    for (var dy: i32 = -1; dy <= 1; dy++) {
        let cy = u32(clamp(i32(y) + dy, 0, i32(dims.height) - 1));
        let row = kernel_weights.weights[dy + 1];
        for (var dx: i32 = -1; dx <= 1; dx++) {
            let cx = u32(clamp(i32(x) + dx, 0, i32(dims.width) - 1));
            let p = input_rgba[cy * dims.width + cx];
            acc += vec3<f32>(f32(p & 0xFFu), f32((p >> 8u) & 0xFFu), f32((p >> 16u) & 0xFFu)) * row[dx + 1];
        }
    }
    let a = input_rgba[y * dims.width + x] >> 24u;
    let c = vec3<u32>(clamp(acc, vec3<f32>(0.0), vec3<f32>(255.0)));
    output_rgba[y * dims.width + x] = c.x | (c.y << 8u) | (c.z << 16u) | (a << 24u);
}
"#;
fn bench(mut f: impl FnMut()) -> (f64, f64) {
    for _ in 0..3 { f(); }
    let mut t = Vec::new(); let st = Instant::now();
    while t.len() < 10 || st.elapsed().as_millis() < 400 { let s = Instant::now(); f(); t.push(s.elapsed().as_secs_f64() * 1e3); }
    t.sort_by(|a, b| a.partial_cmp(b).unwrap());
    let m = t[t.len() / 2];
    let mut d: Vec<f64> = t.iter().map(|x| (x - m).abs()).collect(); d.sort_by(|a, b| a.partial_cmp(b).unwrap());
    (m, d[d.len() / 2])
}
fn main() {
    let instance = wgpu::Instance::default();
    let adapter = pollster::block_on(instance.request_adapter(&wgpu::RequestAdapterOptions { power_preference: wgpu::PowerPreference::HighPerformance, compatible_surface: None, force_fallback_adapter: false, apply_limit_buckets: false })).unwrap();
    let (device, queue) = pollster::block_on(adapter.request_device(&wgpu::DeviceDescriptor::default())).unwrap();
    let module = device.create_shader_module(wgpu::ShaderModuleDescriptor { label: None, source: wgpu::ShaderSource::Wgsl((if std::env::var("LEAN").is_ok() { LEAN_WGSL } else { oxipix::gpu::shaders::CONVOLUTION_2D_WGSL }).into()) });
    let pipeline = device.create_compute_pipeline(&wgpu::ComputePipelineDescriptor { label: None, layout: None, module: &module, entry_point: Some("main"), compilation_options: Default::default(), cache: None });
    let kernel: [f32; 9] = [1.0 / 16.0, 2.0 / 16.0, 1.0 / 16.0, 2.0 / 16.0, 4.0 / 16.0, 2.0 / 16.0, 1.0 / 16.0, 2.0 / 16.0, 1.0 / 16.0];
    let k: [f32; 12] = [kernel[0], kernel[1], kernel[2], 0.0, kernel[3], kernel[4], kernel[5], 0.0, kernel[6], kernel[7], kernel[8], 0.0];
    println!("adapter: {} shader: {}", adapter.get_info().name, if std::env::var("LEAN").is_ok() {"lean"} else {"oxipix"});
    for (w, h) in [(1920u32, 1080u32), (3840, 2160)] {
        let rgba: Vec<u8> = (0..w * h * 4).map(|i| ((i.wrapping_mul(2654435761)) >> 13) as u8).collect();
        let len = rgba.len() as u64;
        let input = device.create_buffer_init(&wgpu::util::BufferInitDescriptor { label: None, contents: &rgba, usage: wgpu::BufferUsages::STORAGE | wgpu::BufferUsages::COPY_DST });
        let kb = device.create_buffer_init(&wgpu::util::BufferInitDescriptor { label: None, contents: bytemuck::cast_slice(&k), usage: wgpu::BufferUsages::UNIFORM });
        let db = device.create_buffer_init(&wgpu::util::BufferInitDescriptor { label: None, contents: bytemuck::cast_slice(&[w, h, 0u32, 0u32]), usage: wgpu::BufferUsages::UNIFORM });
        let out = device.create_buffer(&wgpu::BufferDescriptor { label: None, size: len, usage: wgpu::BufferUsages::STORAGE | wgpu::BufferUsages::COPY_SRC, mapped_at_creation: false });
        let staging = device.create_buffer(&wgpu::BufferDescriptor { label: None, size: len, usage: wgpu::BufferUsages::MAP_READ | wgpu::BufferUsages::COPY_DST, mapped_at_creation: false });
        let bg = device.create_bind_group(&wgpu::BindGroupDescriptor { label: None, layout: &pipeline.get_bind_group_layout(0), entries: &[
            wgpu::BindGroupEntry { binding: 0, resource: input.as_entire_binding() },
            wgpu::BindGroupEntry { binding: 1, resource: kb.as_entire_binding() },
            wgpu::BindGroupEntry { binding: 2, resource: db.as_entire_binding() },
            wgpu::BindGroupEntry { binding: 3, resource: out.as_entire_binding() }] });
        let dispatch = |copy: bool| {
            let mut enc = device.create_command_encoder(&Default::default());
            { let mut p = enc.begin_compute_pass(&Default::default()); p.set_pipeline(&pipeline); p.set_bind_group(0, &bg, &[]); p.dispatch_workgroups(w.div_ceil(16), h.div_ceil(16), 1); }
            if copy { enc.copy_buffer_to_buffer(&out, 0, &staging, 0, len); }
            queue.submit(Some(enc.finish()));
        };
        let read = || -> Vec<u8> {
            let s = staging.slice(..); s.map_async(wgpu::MapMode::Read, |_| {});
            device.poll(wgpu::PollType::wait_indefinitely()).unwrap();
            let v = s.get_mapped_range().unwrap().to_vec(); staging.unmap(); v
        };
        let (res, res_mad) = bench(|| { dispatch(false); device.poll(wgpu::PollType::wait_indefinitely()).unwrap(); });
        let (rt, rt_mad) = bench(|| { queue.write_buffer(&input, 0, &rgba); dispatch(true); std::hint::black_box(read()); });
        dispatch(true); let gpu = read();
        let cpu = oxipix::gpu::cpu::convolve3x3(&rgba, w, h, &kernel);
        println!("{w}x{h}: wgsl resident {res:.3} ms (mad {res_mad:.3}) | wgsl pooled roundtrip {rt:.3} ms (mad {rt_mad:.3}) | exact={}", gpu == cpu);
    }
}
